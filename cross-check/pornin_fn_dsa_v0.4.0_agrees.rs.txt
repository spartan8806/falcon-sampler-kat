    // Exact f64 -> FLR. v = mant * 2^(exp-1075) for a normal double.
    fn flr(v: f64) -> FLR {
        let bits = v.to_bits();
        let neg = (bits >> 63) != 0;
        let exp = ((bits >> 52) & 0x7ff) as i32;
        assert!(exp != 0 && exp != 0x7ff, "not a normal double");
        let mant = ((bits & ((1u64 << 52) - 1)) | (1u64 << 52)) as i64;
        FLR::scaled(if neg { -mant } else { mant }, exp - 1075)
    }
