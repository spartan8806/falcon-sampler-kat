    /// The falcon-sampler-kat vectors against the UNDER-PROVISIONED sampler (v0.1.3, FixedPoint64).
    /// Expected answers come from PQClean. Disagreement here is the POINT: it is the vectors
    /// catching the real historical defect in the real implementation, not a synthetic degradation.
    #[test]
    fn falcon_sampler_kat_catches_the_defect() {
        let vectors: [(f64, f64, [u8; 7], bool, &str); 20] = [
        (0.05f64, 0.7099076094444444f64, [0xac, 0xdf, 0x7a, 0x6b, 0x99, 0x4c, 0x3e], true, "Falcon-512"),
        (0.05f64, 0.7099076094444444f64, [0xac, 0xdf, 0x7a, 0x6b, 0x9a, 0xa5, 0xfd], false, "Falcon-512"),
        (0.2000001676183812f64, 0.7099076094444444f64, [0x94, 0xcb, 0x09, 0x65, 0x94, 0x0a, 0xb9], true, "Falcon-512"),
        (0.2000001676183812f64, 0.7099076094444444f64, [0x94, 0xcb, 0x09, 0x65, 0x95, 0x34, 0x4f], false, "Falcon-512"),
        (0.35f64, 0.7099076094444444f64, [0x80, 0x11, 0x43, 0xbe, 0x0d, 0xe7, 0x0f], true, "Falcon-512"),
        (0.35f64, 0.7099076094444444f64, [0x80, 0x11, 0x43, 0xbe, 0x0e, 0xe7, 0x31], false, "Falcon-512"),
        (0.5f64, 0.7099076094444444f64, [0x6e, 0x3a, 0x89, 0xec, 0x9b, 0x00, 0xcf], true, "Falcon-512"),
        (0.5f64, 0.7099076094444444f64, [0x6e, 0x3a, 0x89, 0xec, 0x9b, 0xdd, 0x44], false, "Falcon-512"),
        (0.65f64, 0.7099076094444444f64, [0x5e, 0xdf, 0xeb, 0xe0, 0xd5, 0x6f, 0xef], true, "Falcon-512"),
        (0.65f64, 0.7099076094444444f64, [0x5e, 0xdf, 0xeb, 0xe0, 0xd6, 0x2d, 0xaf], false, "Falcon-512"),
        (0.05f64, 0.7212668522222222f64, [0xaf, 0xa3, 0x9c, 0x5d, 0xaa, 0x93, 0xfa], true, "Falcon-1024"),
        (0.05f64, 0.7212668522222222f64, [0xaf, 0xa3, 0x9c, 0x5d, 0xab, 0xf3, 0x42], false, "Falcon-1024"),
        (0.2000001676183812f64, 0.7212668522222222f64, [0x97, 0x2c, 0x88, 0x2f, 0xea, 0xc1, 0x45], true, "Falcon-1024"),
        (0.2000001676183812f64, 0.7212668522222222f64, [0x97, 0x2c, 0x88, 0x2f, 0xeb, 0xef, 0x9e], false, "Falcon-1024"),
        (0.35f64, 0.7212668522222222f64, [0x82, 0x1d, 0xdc, 0xb5, 0xf4, 0x92, 0x07], true, "Falcon-1024"),
        (0.35f64, 0.7212668522222222f64, [0x82, 0x1d, 0xdc, 0xb5, 0xf5, 0x96, 0x43], false, "Falcon-1024"),
        (0.5f64, 0.7212668522222222f64, [0x6f, 0xfe, 0x10, 0x65, 0x68, 0x23, 0xc4], true, "Falcon-1024"),
        (0.5f64, 0.7212668522222222f64, [0x6f, 0xfe, 0x10, 0x65, 0x69, 0x03, 0xc1], false, "Falcon-1024"),
        (0.65f64, 0.7212668522222222f64, [0x60, 0x64, 0x8d, 0x84, 0xf7, 0x94, 0x58], true, "Falcon-1024"),
        (0.65f64, 0.7212668522222222f64, [0x60, 0x64, 0x8d, 0x84, 0xf8, 0x55, 0x21], false, "Falcon-1024"),
        ];
        let mut caught = 0;
        for (x, ccs, bytes, expect, pset) in vectors.iter() {
            let got = ber_exp(FixedPoint64::from(*x), FixedPoint64::from(*ccs), *bytes);
            if got != *expect {
                caught += 1;
                println!("CAUGHT  {}  x={}  reference={}  v0.1.3={}", pset, x, expect, got);
            }
        }
        println!("
{} of {} vectors detect the under-provisioned sampler", caught, vectors.len());

        // ASSERT, do not merely print. Without this the test reports `ok` whatever `caught` is --
        // including 0 -- so the arm carrying the actual historical defect could not fail. This
        // repository argues that a check unable to go red is decoration; that has to apply to its
        // own cross-check first. (Found by adversarial review, 2026-08-14.)
        //
        // 10 and not 20, and the number is the point: each x contributes one accept-direction and
        // one reject-direction vector, an implementation errs one way at a given x, so exactly one
        // of each pair can flip. 20 here would mean something had gone wrong.
        assert_eq!(
            caught, 10,
            "expected exactly 10 of 20 vectors to catch the v0.1.3 sampler (one per parameter \
             point, in that implementation's error direction); got {}", caught
        );
    }
